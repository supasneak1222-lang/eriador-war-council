-- Stand-ins for the Tabletop Simulator API, so the War Council's rules run headless (in the new app's engine).
-- Messages the War Council would show in TTS are collected in OUTBOX for the app to display.
OUTBOX = {}
local function out(kind, msg, target) OUTBOX[#OUTBOX + 1] = { kind = kind, msg = tostring(msg), to = target } end
function broadcastToAll(m) out('broadcast', m) end
function broadcastToColor(m, color) out('broadcast', m, color) end
function printToColor(m, color) out('print', m, color) end
function printToAll(m) out('print', m) end
function log() end

function UI.setAttribute() end

-- timers run immediately (the app drives time itself)
Wait = {}
function Wait.frames(f) f(); return 0 end
function Wait.time(f, t, reps) if reps == nil or reps == 1 or reps == 0 then f() end; return 0 end
function Wait.condition(f, c) if c() then f() end; return 0 end
function Wait.stop() end

-- the table: no physical objects; the map board answers geometry questions from the app

  if fn == "territoryAt" then return TERRITORY_AT and TERRITORY_AT(p) or nil end

end
